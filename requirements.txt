# =============================================================================
# Eagle Dashboard — dependencies
#
# Django is the only package the APPLICATION code imports. Everything else runs
# on the Python standard library on purpose:
#   WhatsApp Cloud API  -> urllib          (dashboard/whatsapp.py)
#   Claude API          -> urllib          (dashboard/ai.py)
#   Incoming e-mail     -> imaplib, email  (management/commands/fetch_emails.py)
#   Outgoing e-mail     -> django.core.mail over SMTP (dashboard/mailer.py)
#   .docx text extract  -> zipfile         (dashboard/ai.py)
#   Webhook signatures  -> hmac, hashlib   (dashboard/whatsapp.py)
#
# The packages below Django exist only to run it on a host (gunicorn, whitenoise,
# daphne), to reach the database (psycopg) or to push live events (channels). Do not add one without a reason this
# file can state in a line.
#
# Install:  pip install -r requirements.txt
# =============================================================================

Django==5.2.17

# PostgreSQL driver — needed only when DATABASE_URL is set (Neon on the server).
# Harmless locally: with no DATABASE_URL the project falls back to SQLite and
# never imports it.
psycopg[binary]==3.2.3


# Production serving on Render. daphne (below) is what runs there now; gunicorn is
# the WSGI fallback if daphne ever has to be backed out (set the start command to
# `gunicorn Core.wsgi:application`). whitenoise hands out /static/ because a PaaS
# has no web server in front (settings.py turns it on only when DEBUG is off, so
# local work never imports it).
gunicorn==23.0.0
whitenoise==6.8.2

# Live pushes over WebSocket (dashboard/realtime.py). channels is the Django side,
# daphne the ASGI server that runs it on Render, channels-redis the layer that lets
# the web process and the worker talk to each other.
channels==4.3.2
daphne==4.2.3
channels-redis==4.3.0
